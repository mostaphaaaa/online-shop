@using ShopStock.Domain.ViewModels.Common
@model PagingViewModel
<ul class="pagination justify-content-center">

    @if (Model.StartPage < Model.PageId)
    {
        <li class="page-item">
            <a class="page-link cursor-pointer" href="@ViewBag.Url?PageId=@(Model.PageId - 1)">
                <i class="fa fa-chevron-right"></i>
            </a>
        </li>
    }

    @for (int i = Model.StartPage; i <= Model.EndPage; i++)
    {
        <li class="page-item @((Model.PageId == i) ? "active" : "")">
            <a class="page-link " href="@ViewBag.Url?PageId=@i">@i</a>
        </li>
        @* @if (Model.PageId == i)
        {
            <li class="page-item active">
                <a class="page-link" href="@ViewBag.Url?PageId=@i">@i</a>
            </li>
        }
        else
        {
            <li class="page-item ">
                <a class="page-link" href="@ViewBag.Url?PageId=@i">@i</a>
            </li>
        } *@
    }
    @if (Model.EndPage > Model.PageId)
    {
        <li class="page-item">
            <a class="page-link cursor-pointer" href="@ViewBag.Url?PageId=@(Model.PageId + 1)">
                <i class="fa fa-chevron-left"></i>
            </a>
        </li>
    }
</ul>
